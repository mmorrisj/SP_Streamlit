# PyTorch CPU-only (~500MB)
# Installed from pytorch CPU index, not PyPI
# Must match version downloaded by production-build.sh (>=2.6.0 to avoid CVE-2025-32434)
torch>=2.6.0

# Sentence-transformers + transitive deps (~300MB)
# Pulls: transformers, scipy, scikit-learn, tokenizers, safetensors, nltk, Pillow
# Cap majors: nomic-embed-text-v1.5's trust_remote_code model targets the
# transformers 4.x API. transformers 5.x mis-maps the checkpoint and loads
# RANDOM weights (encoder.encoder.* key mismatch) -> garbage embeddings.
transformers>=4.44,<5
sentence-transformers>=3.3.1,<4
# huggingface_hub must stay <1.0 to match the transformers 4.x line (5.x +
# hub 1.0 broke nomic's trust_remote_code loading). transformers pins a
# compatible hub (~0.36); offline trust_remote_code still works because
# bake_models.py rewrites nomic's auto_map to local module paths.
huggingface_hub<1.0

# LangChain HuggingFace wrapper (for RAG embedding)
# Pulls: langchain-core
langchain-huggingface>=1.0

# LangChain PostgreSQL vector store (migrated from langchain-community)
# Pulls: langchain-core, psycopg
langchain-postgres>=0.0.16,<0.1.0
